výběr jen pro Founder/Council). Nemění nic z existující logiky odvodu
("Zaplatit a podepsat"), jen přidává dva nové bloky POD `<div id="rf-list">`
v panelu Reserve Fund (uvnitř `panel-ucet`).
══════════════════════════════════════════════════════════════════════

2) Do <script> sekce (kdekoliv u ostatních Reserve Fund funkcí, poblíž
   `payReserveFund` / `loadReserveFund`) přidej:

async function rfDeposit(){
  const amount = document.getElementById('rf-deposit-amount').value;
  if(!amount || parseFloat(amount) <= 0) return showToast('Vyplň platnou částku', true);
  const r = await post('/api/reserve-fund/deposit', { amount });
  if(r.ok){ showToast('Vklad do Reserve Fondu zaznamenán'); document.getElementById('rf-deposit-amount').value=''; flashActivePanel(); loadReserveFund(); }
  else showToast(r.error, true);
}
window.rfDeposit = rfDeposit;

async function rfWithdraw(){
  const amount = document.getElementById('rf-withdraw-amount').value;
  const poznamka = document.getElementById('rf-withdraw-reason').value.trim();
  if(!amount || parseFloat(amount) <= 0) return showToast('Vyplň platnou částku', true);
  if(!poznamka) return showToast('Vyplň důvod výběru', true);
  showModal(
    'Výběr z Reserve Fondu',
    'Potvrzením vybereš tuto částku ze samostatného účtu Reserve Fondu.',
    [['Částka', '$'+amount], ['Důvod', poznamka]],
    async () => {
      const r = await post('/api/reserve-fund/withdraw', { amount, poznamka });
      if(r.ok){ showToast('Výběr zaznamenán'); document.getElementById('rf-withdraw-amount').value=''; document.getElementById('rf-withdraw-reason').value=''; flashActivePanel(); loadReserveFund(); }
      else showToast(r.error, true);
    }
  );
}
window.rfWithdraw = rfWithdraw;

(`post()` a `showModal()` už v souboru existují — žádné další závislosti.)
